    const profSnap = await db.collection('site').doc('profile').get();
    if(profSnap.exists) profile = { ...profile, ...profSnap.data() };
  }catch(e){ console.error('Could not load profile', e); }
  try{
    const postsSnap = await db.collection('site').doc('posts').get();
    if(postsSnap.exists && Array.isArray(postsSnap.data().items)) posts = postsSnap.data().items;
  }catch(e){ console.error('Could not load posts', e); posts = []; }
}
async function saveProfileData(){
  try{
    await db.collection('site').doc('profile').set(profile);
  }catch(e){ alert('Could not save profile: ' + e.message); }
}
async function savePostsData(){
  try{
    await db.collection('site').doc('posts').set({ items: posts });
  }catch(e){ alert('Could not save writing: ' + e.message); }
}

/* ---------- helpers ---------- */
function esc(s){ return (s||'').replace(/[&<>"']/g, c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }
function formatDate(iso){
  if(!iso) return '';
  const d = new Date(iso + 'T00:00:00');
  return d.toLocaleDateString('en-US', { month:'long', day:'numeric', year:'numeric' });
}
function readingTime(body){
  const words = (body||'').trim().split(/\s+/).filter(Boolean).length;
  return Math.max(1, Math.round(words/200));
}
function initials(name){
  return (name||'').split(' ').filter(Boolean).slice(0,2).map(w=>w[0].toUpperCase()).join('');
}

/* ---------- render ---------- */
function renderProfile(){
  document.getElementById('brand-name').textContent = profile.name;
  document.getElementById('hero-name').textContent = profile.name;
  document.getElementById('hero-tagline').textContent = profile.tagline;
  document.getElementById('about-body').textContent = profile.about;
  document.getElementById('footer-name').textContent = profile.name;
  const avatar = document.getElementById('avatar');
  avatar.innerHTML = profile.avatarUrl
    ? `<img src="${esc(profile.avatarUrl)}" alt="${esc(profile.name)}">`
    : initials(profile.name);
  const socialRow = document.getElementById('social-row');
  socialRow.innerHTML = (profile.links||[]).map(l=>`<a href="${esc(l.url)}" target="_blank" rel="noopener">${esc(l.label)}</a>`).join('');
  const contactRow = document.getElementById('contact-row');
  contactRow.innerHTML = profile.contactEmail
    ? `<a class="contact-pill" href="mailto:${esc(profile.contactEmail)}">For publishing &amp; research inquiries — ${esc(profile.contactEmail)}</a>`
    : '';

  const counts = { Poem:0, Story:0, Blog:0, Thesis:0, Other:0 };
  posts.forEach(p => { counts[p.type] = (counts[p.type]||0) + 1; });
  const statsRow = document.getElementById('stats-row');
  const parts = [];
  if(counts.Poem) parts.push(['Poems', counts.Poem]);
  if(counts.Story) parts.push(['Stories', counts.Story]);
  if(counts.Blog) parts.push(['Blog posts', counts.Blog]);
  if(counts.Thesis) parts.push(['Thesis work', counts.Thesis]);
  if(counts.Other) parts.push(['Other', counts.Other]);
  statsRow.innerHTML = parts.map(([l,n])=>`<div class="stat"><div class="n">${n}</div><div class="l">${esc(l)}</div></div>`).join('');
}

function renderFilters(){
  const types = ['All', ...new Set(posts.map(p=>p.type))];
  document.getElementById('filters').innerHTML = types.map(t=>
    `<button class="filter-btn ${t===activeFilter?'active':''}" data-t="${esc(t)}">${esc(t)}</button>`
  ).join('');
  document.querySelectorAll('.filter-btn').forEach(btn=>{
    btn.addEventListener('click', ()=>{ activeFilter = btn.dataset.t; renderFilters(); renderList(); });
  });
}

function renderList(){
  const sorted = [...posts].sort((a,b)=> new Date(b.date) - new Date(a.date));
  const shown = activeFilter==='All' ? sorted : sorted.filter(p=>p.type===activeFilter);
  document.getElementById('post-count').textContent = `${shown.length} piece${shown.length!==1?'s':''}`;
  const listEl = document.getElementById('post-list');

  if(shown.length===0){
    listEl.innerHTML = `<div class="empty-state">${posts.length===0 ? "Nothing published yet. Click “Write” above to add the first piece." : "Nothing in this category yet."}</div>`;
    return;
  }

  listEl.innerHTML = shown.map(p=>{
    const tags = (p.tags||[]).map(t=>`<span class="tag">${esc(t)}</span>`).join('');
    return `
    <article class="post-card">
      <div class="post-clickable" data-id="${p.id}">
        <div class="post-meta"><span class="type-badge">${esc(p.type)}</span><span>${formatDate(p.date)} · ${readingTime(p.body)} min read</span></div>
        <h3>${esc(p.title)}</h3>
        ${p.excerpt ? `<p class="post-excerpt">${esc(p.excerpt)}</p>` : ''}
        <div class="post-tags">${tags}</div>
      </div>
      ${isAuthor ? `<div class="card-actions">
        <button class="icon-btn" onclick="openPostForm('${p.id}')">Edit</button>
        <button class="icon-btn danger" onclick="deletePost('${p.id}')">Delete</button>
      </div>` : ''}
    </article>`;
  }).join('');

  listEl.querySelectorAll('.post-clickable').forEach(el=>{
    el.addEventListener('click', ()=> openPost(el.dataset.id));
  });
}

function renderAuthorUI(){
  document.getElementById('author-status').innerHTML = isAuthor
    ? `<span class="author-pill">${authorRole === 'admin' ? 'Admin mode' : 'Author mode'}</span>` : '';
  document.getElementById('write-toggle').style.display = isAuthor ? 'inline-block' : 'none';
  renderList();
}

/* ---------- post detail ---------- */
function openPost(id){
  const p = posts.find(x=>x.id===id);
  if(!p) return;
  document.getElementById('detail-cover-wrap').innerHTML = p.coverImageUrl ? `<img class="detail-cover" src="${esc(p.coverImageUrl)}" alt="">` : '';
  document.getElementById('detail-meta').textContent = `${p.type} · ${formatDate(p.date)} · ${readingTime(p.body)} min read`;
  document.getElementById('detail-title').textContent = p.title;
  document.getElementById('detail-tags').innerHTML = (p.tags||[]).map(t=>`<span class="tag">${esc(t)}</span>`).join('');
  document.getElementById('detail-body').textContent = p.body;
  document.getElementById('post-overlay').classList.add('open');
  window.scrollTo(0,0);
  document.body.style.overflow = 'hidden';
}
function closePost(){
  document.getElementById('post-overlay').classList.remove('open');
  document.body.style.overflow = '';
}

/* ---------- modals ---------- */
function openModal(id){ document.getElementById(id).classList.add('open'); }
function closeModal(id){ document.getElementById(id).classList.remove('open'); }

document.getElementById('write-toggle').addEventListener('click', ()=>{
  if(isAuthor){ openModal('author-menu-backdrop'); }
  else{ openLoginPrompt(); }
});
function openLoginPrompt(){
  document.getElementById('passcode-input').value = '';
  document.getElementById('passcode-error').classList.remove('show');
  openModal('passcode-backdrop');
}
function checkPasscode(){
  const val = document.getElementById('passcode-input').value;
  if(val === AUTHOR_PASSCODE || val === ADMIN_PASSCODE){
    isAuthor = true;
    authorRole = (val === ADMIN_PASSCODE) ? 'admin' : 'author';
    closeModal('passcode-backdrop');
    renderAuthorUI();
    openModal('author-menu-backdrop');
  }else{
    document.getElementById('passcode-error').classList.add('show');
  }
}
function lockAuthor(){
  isAuthor = false;
  authorRole = null;
  closeModal('author-menu-backdrop');
  renderAuthorUI();
}
document.getElementById('passcode-input').addEventListener('keydown', e=>{ if(e.key==='Enter') checkPasscode(); });

/* hidden entry points — nothing about these appears on the page */
document.getElementById('avatar').addEventListener('click', ()=>{
  avatarClicks++;
  clearTimeout(avatarClickTimer);
  avatarClickTimer = setTimeout(()=>{ avatarClicks = 0; }, 2500);
  if(avatarClicks >= 5){
    avatarClicks = 0;
    if(!isAuthor) openLoginPrompt();
  }
});

/* ---------- post form ---------- */
function openPostForm(id){
  editingId = id || null;
  document.getElementById('post-form-error').classList.remove('show');
  if(id){
    const p = posts.find(x=>x.id===id);
    document.getElementById('post-form-title').textContent = 'Edit piece';
    document.getElementById('pf-type').value = p.type;
    document.getElementById('pf-date').value = p.date;
    document.getElementById('pf-title').value = p.title;
    document.getElementById('pf-tags').value = (p.tags||[]).join(', ');
    document.getElementById('pf-cover').value = p.coverImageUrl || '';
    document.getElementById('pf-excerpt').value = p.excerpt || '';
    document.getElementById('pf-body').value = p.body;
  }else{
    document.getElementById('post-form-title').textContent = 'New piece of writing';
    document.getElementById('pf-type').value = 'Poem';
    document.getElementById('pf-date').value = new Date().toISOString().slice(0,10);
    document.getElementById('pf-title').value = '';
    document.getElementById('pf-tags').value = '';
    document.getElementById('pf-cover').value = '';
    document.getElementById('pf-excerpt').value = '';
    document.getElementById('pf-body').value = '';
  }
  openModal('post-form-backdrop');
}
async function savePost(){
  const title = document.getElementById('pf-title').value.trim();
  const body = document.getElementById('pf-body').value.trim();
  if(!title || !body){
    document.getElementById('post-form-error').classList.add('show');
    return;
  }
  const data = {
    type: document.getElementById('pf-type').value,
    date: document.getElementById('pf-date').value || new Date().toISOString().slice(0,10),
    title,
    tags: document.getElementById('pf-tags').value.split(',').map(s=>s.trim()).filter(Boolean),
    coverImageUrl: document.getElementById('pf-cover').value.trim(),
    excerpt: document.getElementById('pf-excerpt').value.trim(),
    body
  };
  if(editingId){
    const idx = posts.findIndex(x=>x.id===editingId);
    posts[idx] = { ...posts[idx], ...data };
  }else{
    posts.push({ id: 'p_' + Date.now() + '_' + Math.random().toString(36).slice(2,7), ...data });
  }
  await savePostsData();
  closeModal('post-form-backdrop');
  renderProfile();
  renderFilters();
  renderList();
}
async function deletePost(id){
  if(!confirm('Delete this piece? This cannot be undone.')) return;
  posts = posts.filter(p=>p.id!==id);
  await savePostsData();
  renderProfile();
  renderFilters();
  renderList();
}

/* ---------- profile form ---------- */
function addLinkRow(label, url){
  const wrap = document.getElementById('pr-links');
  const row = document.createElement('div');
  row.className = 'link-row';
  row.innerHTML = `<input type="text" placeholder="Label (e.g. Instagram)" class="pr-link-label" value="${esc(label)}">
                    <input type="text" placeholder="URL" class="pr-link-url" value="${esc(url)}">
                    <button class="icon-btn danger" type="button" onclick="this.parentElement.remove()">Remove</button>`;
  wrap.appendChild(row);
}
function openProfileForm(){
  document.getElementById('pr-name').value = profile.name;
  document.getElementById('pr-tagline').value = profile.tagline;
  document.getElementById('pr-avatar').value = profile.avatarUrl;
  document.getElementById('pr-contact-email').value = profile.contactEmail || '';
  document.getElementById('pr-about').value = profile.about;
  document.getElementById('pr-links').innerHTML = '';
  (profile.links||[]).forEach(l=>addLinkRow(l.label, l.url));
  openModal('profile-form-backdrop');
}
async function saveProfile(){
  const links = [...document.querySelectorAll('#pr-links .link-row')].map(row=>({
    label: row.querySelector('.pr-link-label').value.trim(),
    url: row.querySelector('.pr-link-url').value.trim()
  })).filter(l=>l.label && l.url);
  profile = {
    name: document.getElementById('pr-name').value.trim() || profile.name,
    tagline: document.getElementById('pr-tagline').value.trim(),
    avatarUrl: document.getElementById('pr-avatar').value.trim(),
    contactEmail: document.getElementById('pr-contact-email').value.trim(),
    about: document.getElementById('pr-about').value.trim(),
    links
  };
  await saveProfileData();
  closeModal('profile-form-backdrop');
  renderProfile();
}

(async function init(){
  if(firebaseConfig.apiKey.startsWith('PASTE_YOUR')){
    document.getElementById('loading').innerHTML =
      'This site isn\'t connected to a database yet. Open this file and follow the "CONFIG #1 — Firebase" ' +
      'instructions near the top of the &lt;script&gt; section, then re-upload it.';
    return;
  }
  try{
    firebase.initializeApp(firebaseConfig);
    db = firebase.firestore();
  }catch(e){
    document.getElementById('loading').textContent = 'Could not connect to the database: ' + e.message;
    return;
  }
  await loadData();
  renderProfile();
  renderFilters();
  renderList();
  document.getElementById('loading').style.display = 'none';
  document.getElementById('app').style.display = 'block';
  if(location.hash === '#write'){ openLoginPrompt(); }
})();
</script>
